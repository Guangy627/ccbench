#!/usr/bin/env python3
"""
数据分析脚本 - Boston Housing 和 Titanic 数据集
包含异常值检测、特征工程、数据可视化和清洗功能
"""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats
import warnings
warnings.filterwarnings('ignore')

# 设置中文字体
plt.rcParams['font.sans-serif'] = ['DejaVu Sans', 'Arial Unicode MS', 'SimHei']
plt.rcParams['axes.unicode_minus'] = False

def log_progress(message):
    """打印进度日志"""
    print(f"[INFO] {message}")

def analyze_boston_housing():
    """分析 Boston Housing 数据集"""
    log_progress("开始分析 Boston Housing 数据集...")

    # 读取数据
    log_progress("读取 Boston Housing 数据...")
    boston_df = pd.read_csv('/app/projects/ml_datasets/BostonHousing.txt')

    print(f"数据集形状: {boston_df.shape}")
    print(f"列名: {list(boston_df.columns)}")
    print(f"数据类型:\n{boston_df.dtypes}")

    # 1. 检查 medv 列异常值（IQR 方法）
    log_progress("检查 medv 列异常值...")
    Q1 = boston_df['medv'].quantile(0.25)
    Q3 = boston_df['medv'].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR

    outliers = boston_df[(boston_df['medv'] < lower_bound) | (boston_df['medv'] > upper_bound)]
    print(f"medv 统计信息:")
    print(f"Q1: {Q1:.2f}, Q3: {Q3:.2f}, IQR: {IQR:.2f}")
    print(f"异常值范围: < {lower_bound:.2f} 或 > {upper_bound:.2f}")
    print(f"发现 {len(outliers)} 个异常值")

    # 2. 找出与 medv 相关性最强的 3 个特征
    log_progress("计算特征相关性...")
    correlation = boston_df.corr()['medv'].abs().sort_values(ascending=False)
    top3_features = correlation.iloc[1:4]  # 排除自身
    print(f"\n与 medv 相关性最强的 3 个特征:")
    for feature, corr in top3_features.items():
        print(f"{feature}: {corr:.4f}")

    # 3. 创建 rm_rounded 列并计算平均房价
    log_progress("创建 rm_rounded 列...")
    boston_df['rm_rounded'] = boston_df['rm'].round()
    avg_price_by_rooms = boston_df.groupby('rm_rounded')['medv'].mean().sort_index()
    print(f"\n每组房间数的平均房价:")
    for rooms, price in avg_price_by_rooms.items():
        print(f"{int(rooms)} 间房: ${price:.2f}K")

    # 4. 绘制 rm 与 medv 的散点图和趋势线
    log_progress("绘制散点图和趋势线...")
    plt.figure(figsize=(10, 6))
    plt.scatter(boston_df['rm'], boston_df['medv'], alpha=0.6, color='blue')

    # 添加趋势线
    z = np.polyfit(boston_df['rm'], boston_df['medv'], 1)
    p = np.poly1d(z)
    plt.plot(boston_df['rm'], p(boston_df['rm']), "r--", alpha=0.8, linewidth=2)

    plt.xlabel('房间数 (rm)')
    plt.ylabel('房价 (medv) - $1000s')
    plt.title('房间数与房价关系散点图')
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.savefig('/app/projects/boston_rm_medv_scatter.png', dpi=300, bbox_inches='tight')
    plt.close()

    # 5. 保存清洗后的数据（去除异常值）
    log_progress("保存清洗后的数据...")
    boston_clean = boston_df[(boston_df['medv'] >= lower_bound) & (boston_df['medv'] <= upper_bound)]
    boston_clean.to_csv('/app/projects/boston_clean.csv', index=False)
    print(f"清洗前数据: {len(boston_df)} 行")
    print(f"清洗后数据: {len(boston_clean)} 行")
    print(f"已保存到 boston_clean.csv")

    return {
        'outliers_count': len(outliers),
        'top3_features': dict(top3_features),
        'avg_price_by_rooms': dict(avg_price_by_rooms),
        'original_rows': len(boston_df),
        'cleaned_rows': len(boston_clean)
    }

def analyze_titanic():
    """分析 Titanic 数据集"""
    log_progress("开始分析 Titanic 数据集...")

    # 读取数据
    log_progress("读取 Titanic 数据...")
    titanic_df = pd.read_csv('/app/projects/ml_datasets/titanic.csv')

    print(f"数据集形状: {titanic_df.shape}")
    print(f"列名: {list(titanic_df.columns)}")

    # 1. 检查缺失值
    log_progress("检查缺失值...")
    missing_values = titanic_df.isnull().sum()
    print(f"\n缺失值统计:")
    for col, missing in missing_values.items():
        if missing > 0:
            print(f"{col}: {missing} ({missing/len(titanic_df)*100:.1f}%)")

    # 填充 age 列（用中位数）
    log_progress("填充 age 列缺失值...")
    age_median = titanic_df['age'].median()
    titanic_df['age'] = titanic_df['age'].fillna(age_median)
    print(f"age 列用中位数 {age_median:.1f} 填充")

    # 填充 deck 列（用众数）
    log_progress("填充 deck 列缺失值...")
    if 'deck' in titanic_df.columns:
        deck_mode = titanic_df['deck'].mode()[0] if not titanic_df['deck'].mode().empty else 'Unknown'
        titanic_df['deck'] = titanic_df['deck'].fillna(deck_mode)
        print(f"deck 列用众数 '{deck_mode}' 填充")

    # 2. 转换为分类变量
    log_progress("转换分类变量...")
    if 'embarked' in titanic_df.columns:
        titanic_df['embarked'] = titanic_df['embarked'].astype('category')
    if 'class' in titanic_df.columns:
        titanic_df['class'] = titanic_df['class'].astype('category')

    # 3. 计算生存率并可视化
    log_progress("计算生存率...")

    # 按性别计算生存率
    survival_by_sex = titanic_df.groupby('sex')['survived'].mean()
    print(f"\n按性别生存率:")
    for sex, rate in survival_by_sex.items():
        print(f"{sex}: {rate:.3f} ({rate*100:.1f}%)")

    # 按舱位等级计算生存率
    if 'class' in titanic_df.columns:
        survival_by_class = titanic_df.groupby('class')['survived'].mean()
        print(f"\n按舱位等级生存率:")
        for cls, rate in survival_by_class.items():
            print(f"{cls}: {rate:.3f} ({rate*100:.1f}%)")

    # 绘制生存率柱状图
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))

    # 性别生存率
    survival_by_sex.plot(kind='bar', ax=axes[0], color=['lightcoral', 'lightblue'])
    axes[0].set_title('按性别生存率')
    axes[0].set_xlabel('性别')
    axes[0].set_ylabel('生存率')
    axes[0].tick_params(axis='x', rotation=0)
    axes[0].grid(True, alpha=0.3)

    # 舱位等级生存率
    if 'class' in titanic_df.columns:
        survival_by_class.plot(kind='bar', ax=axes[1], color=['gold', 'lightgreen', 'lightcoral'])
        axes[1].set_title('按舱位等级生存率')
        axes[1].set_xlabel('舱位等级')
        axes[1].set_ylabel('生存率')
        axes[1].tick_params(axis='x', rotation=0)
        axes[1].grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig('/app/projects/titanic_survival_rates.png', dpi=300, bbox_inches='tight')
    plt.close()

    # 4. 创建 is_alone 特征
    log_progress("创建 is_alone 特征...")
    if 'sibsp' in titanic_df.columns and 'parch' in titanic_df.columns:
        titanic_df['is_alone'] = (titanic_df['sibsp'] + titanic_df['parch']) == 0
        alone_count = titanic_df['is_alone'].sum()
        print(f"独自旅行的乘客: {alone_count} ({alone_count/len(titanic_df)*100:.1f}%)")

    # 5. 保存清洗后的数据
    log_progress("保存清洗后的数据...")
    titanic_df.to_csv('/app/projects/titanic_clean.csv', index=False)
    print(f"已保存到 titanic_clean.csv")

    return {
        'total_passengers': len(titanic_df),
        'survival_by_sex': dict(survival_by_sex),
        'survival_by_class': dict(survival_by_class) if 'class' in titanic_df.columns else {},
        'alone_passengers': alone_count if 'sibsp' in titanic_df.columns and 'parch' in titanic_df.columns else 0
    }
