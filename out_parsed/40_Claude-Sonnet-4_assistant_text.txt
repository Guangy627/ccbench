def load_data():
    """加载Titanic数据集"""
    logger.info("正在加载Titanic数据集...")
    df = pd.read_csv('ml_datasets/titanic.csv')
    logger.info(f"数据集加载完成，形状: {df.shape}")
    return df

def check_missing_values(df):
    """检查缺失值"""
    logger.info("检查缺失值...")
    missing_info = df.isnull().sum()
    missing_percent = (missing_info / len(df)) * 100
    
    missing_df = pd.DataFrame({
        'Missing Count': missing_info,
        'Missing Percentage': missing_percent
    })
    missing_df = missing_df[missing_df['Missing Count'] > 0].sort_values('Missing Count', ascending=False)
    
    logger.info("缺失值统计:")
    for col, row in missing_df.iterrows():
        logger.info(f"  {col}: {row['Missing Count']} ({row['Missing Percentage']:.1f}%)")
    
    return missing_df

def fill_missing_values(df):
    """填充缺失值"""
    logger.info("开始填充缺失值...")
    df_clean = df.copy()
    
    # 填充age列 - 使用均值
    if df_clean['age'].isnull().any():
        age_mean = df_clean['age'].mean()
        df_clean['age'].fillna(age_mean, inplace=True)
        logger.info(f"age列缺失值已用均值填充: {age_mean:.1f}")
    
    # 填充deck列 - 使用众数或删除（这里选择用'Unknown'填充）
    if df_clean['deck'].isnull().any():
        df_clean['deck'].fillna('Unknown', inplace=True)
        logger.info("deck列缺失值已用'Unknown'填充")
    
    # 填充embarked列的少数缺失值
    if df_clean['embarked'].isnull().any():
        embarked_mode = df_clean['embarked'].mode()[0]
        df_clean['embarked'].fillna(embarked_mode, inplace=True)
        logger.info(f"embarked列缺失值已用众数填充: {embarked_mode}")
    
    return df_clean

def convert_to_categorical(df):
    """转换为分类变量"""
    logger.info("转换embarked和class为分类变量...")
    df['embarked'] = df['embarked'].astype('category')
    df['class'] = df['class'].astype('category')
    logger.info("分类变量转换完成")
    return df

def calculate_survival_rates(df):
    """计算生存率"""
    logger.info("计算不同群体的生存率...")
    
    # 按性别计算生存率
    survival_by_sex = df.groupby('sex')['survived'].agg(['count', 'sum', 'mean']).round(3)
    survival_by_sex.columns = ['Total', 'Survived', 'Survival_Rate']
    
    # 按舱位等级计算生存率
    survival_by_class = df.groupby('class')['survived'].agg(['count', 'sum', 'mean']).round(3)
    survival_by_class.columns = ['Total', 'Survived', 'Survival_Rate']
    
    # 按性别和舱位等级计算生存率
    survival_by_sex_class = df.groupby(['sex', 'class'])['survived'].agg(['count', 'sum', 'mean']).round(3)
    survival_by_sex_class.columns = ['Total', 'Survived', 'Survival_Rate']
    
    logger.info("生存率计算完成")
    return survival_by_sex, survival_by_class, survival_by_sex_class

这是Python的条件表达式（三元运算符），用来防止KeyError错误。让我解释一下：

## 问题背景
在HTML报告生成时，我们要显示缺失值的百分比：
```python
{missing_df.loc['age', 'Missing Percentage']:.1f}%
```

## 潜在问题
如果原始数据中某列**没有缺失值**，那么这一列就不会出现在`missing_df`中（因为我们过滤了缺失值为0的列）。这时直接访问`missing_df.loc['age', 'Missing Percentage']`会报KeyError。

## 解决方案
```python
{missing_df.loc['age', 'Missing Percentage']:.1f}% if 'age' in missing_df.index else '0.0%'
```

这个表达式的逻辑是：
- **如果** `'age'` 在 `missing_df.index` 中存在
- **那么** 显示实际的缺失百分比：`{missing_df.loc['age', 'Missing Percentage']:.1f}%`  
- **否则** 显示 `'0.0%'`（表示没有缺失值）

## 实际例子
假设某个数据集中age列没有缺失值：
- 没有条件检查：`missing_df.loc['age', 'Missing Percentage']` → **KeyError**
- 有条件检查：`'age' in missing_df.index` 返回False → 显示 `'0.0%'`

这样确保了HTML报告在任何情况下都能正常生成，不会因为某列没有缺失值而崩溃。
