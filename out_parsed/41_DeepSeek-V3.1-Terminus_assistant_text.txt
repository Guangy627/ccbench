import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import json
import os

# 设置中文字体支持
plt.rcParams['font.sans-serif'] = ['SimHei', 'Arial Unicode MS', 'DejaVu Sans']

# 1. 读取数据并筛选2007年数据
print("正在读取gapminder数据集...")
df = pd.read_csv('/app/projects/ml_datasets/gapminderDataFiveYear.csv')
df_2007 = df[df['year'] == 2007]
print(f"2007年数据包含 {len(df_2007)} 个国家")

# 2. 绘制人均GDP与预期寿命散点图（颜色区分大洲）
print("正在绘制散点图...")
plt.figure(figsize=(12, 8))
sns.scatterplot(data=df_2007, x='gdpPercap', y='lifeExp', hue='continent', s=100, alpha=0.7)
plt.title('2007年各国人均GDP与预期寿命关系（按大洲区分）', fontsize=16)
plt.xlabel('人均GDP (美元)', fontsize=12)
plt.ylabel('预期寿命 (年)', fontsize=12)
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('/app/projects/scatter_plot.png', dpi=300, bbox_inches='tight')
plt.close()

# 3. 计算每个大洲2007年平均人口并用柱状图展示
print("正在计算大洲平均人口...")
continent_pop = df_2007.groupby('continent')['pop'].mean().sort_values(ascending=False)

plt.figure(figsize=(10, 6))
continent_pop.plot(kind='bar', color=['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728', '#9467bd'])
plt.title('2007年各洲平均人口', fontsize=16)
plt.xlabel('大洲', fontsize=12)
plt.ylabel('平均人口', fontsize=12)
plt.xticks(rotation=45)
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('/app/projects/continent_population.png', dpi=300, bbox_inches='tight')
plt.close()

# 4. 找出人口增长最快的国家（1952→2007年）
print("正在计算人口增长最快的国家...")
df_1952 = df[df['year'] == 1952][['country', 'pop']]
df_2007_pop = df_2007[['country', 'pop']]

# 合并1952和2007年数据
pop_growth = pd.merge(df_1952, df_2007_pop, on='country', suffixes=('_1952', '_2007'))
pop_growth['growth_rate'] = (pop_growth['pop_2007'] - pop_growth['pop_1952']) / pop_growth['pop_1952'] * 100
pop_growth['growth_absolute'] = pop_growth['pop_2007'] - pop_growth['pop_1952']

# 按增长率排序
growth_fastest = pop_growth.sort_values('growth_rate', ascending=False).head(10)
growth_absolute_fastest = pop_growth.sort_values('growth_absolute', ascending=False).head(10)

# 5. 创建新列gdp_per_capita（GDP总量/人口）
print("正在计算GDP总量...")
df_2007['gdp_total'] = df_2007['gdpPercap'] * df_2007['pop']

# 6. 将2007年数据导出为JSON文件
print("正在导出JSON文件...")
json_data = df_2007.to_dict('records')
with open('/app/projects/gapminder_2007.json', 'w', encoding='utf-8') as f:
    json.dump(json_data, f, ensure_ascii=False, indent=2)
