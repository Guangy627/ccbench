#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
泰坦尼克号数据集清洗和分析脚本
"""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import logging
from datetime import datetime
import os

# 设置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('titanic_analysis.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

def load_data():
    """加载数据"""
    logger.info("开始加载泰坦尼克号数据集...")
    try:
        df = pd.read_csv('ml_datasets/titanic.csv')
        logger.info(f"数据加载成功，共 {len(df)} 行，{len(df.columns)} 列")
        logger.info(f"数据列名: {list(df.columns)}")
        return df
    except Exception as e:
        logger.error(f"数据加载失败: {e}")
        raise

def check_missing_values(df):
    """检查缺失值"""
    logger.info("开始检查缺失值...")
    missing_info = df.isnull().sum()
    missing_percent = (missing_info / len(df)) * 100
    
    missing_df = pd.DataFrame({
        '缺失数量': missing_info,
        '缺失比例(%)': missing_percent
    }).sort_values('缺失数量', ascending=False)
    
    logger.info("缺失值统计:")
    for col in missing_df.index:
        if missing_df.loc[col, '缺失数量'] > 0:
            logger.info(f"  {col}: {missing_df.loc[col, '缺失数量']} 个缺失值 ({missing_df.loc[col, '缺失比例(%)']:.2f}%)")
    
    return missing_df

def fill_missing_values(df):
    """填充缺失值"""
    logger.info("开始填充缺失值...")
    df_clean = df.copy()
    
    # 填充 age 列（使用中位数）
    if 'age' in df_clean.columns:
        age_median = df_clean['age'].median()
        missing_age = df_clean['age'].isnull().sum()
        df_clean['age'].fillna(age_median, inplace=True)
        logger.info(f"age 列: 使用中位数 {age_median:.1f} 填充了 {missing_age} 个缺失值")
    
    # 填充 deck 列（使用众数或删除）
    if 'deck' in df_clean.columns:
        missing_deck = df_clean['deck'].isnull().sum()
        if missing_deck > 0:
            # 如果缺失值过多，考虑删除该列或使用众数
            if missing_deck / len(df_clean) > 0.5:
                logger.info(f"deck 列缺失值过多 ({missing_deck}/{len(df_clean)}，{missing_deck/len(df_clean)*100:.1f}%)，删除该列")
                df_clean.drop('deck', axis=1, inplace=True)
            else:
                deck_mode = df_clean['deck'].mode()[0] if not df_clean['deck'].mode().empty else 'Unknown'
                df_clean['deck'].fillna(deck_mode, inplace=True)
                logger.info(f"deck 列: 使用众数 '{deck_mode}' 填充了 {missing_deck} 个缺失值")
    
    # 处理 embarked 列的缺失值
    if 'embarked' in df_clean.columns:
        embarked_mode = df_clean['embarked'].mode()[0] if not df_clean['embarked'].mode().empty else 'S'
        missing_embarked = df_clean['embarked'].isnull().sum()
        df_clean['embarked'].fillna(embarked_mode, inplace=True)
        logger.info(f"embarked 列: 使用众数 '{embarked_mode}' 填充了 {missing_embarked} 个缺失值")
    
    return df_clean

def convert_to_categorical(df):
    """转换为分类变量"""
    logger.info("开始转换分类变量...")
    df_cat = df.copy()
    
    # 转换 embarked
    if 'embarked' in df_cat.columns:
        df_cat['embarked'] = df_cat['embarked'].astype('category')
        logger.info(f"embarked 列已转换为分类变量，类别: {list(df_cat['embarked'].cat.categories)}")
    
    # 转换 class (pclass)
    if 'pclass' in df_cat.columns:
        df_cat['pclass'] = df_cat['pclass'].astype('category')
        logger.info(f"pclass 列已转换为分类变量，类别: {list(df_cat['pclass'].cat.categories)}")
    elif 'class' in df_cat.columns:
        df_cat['class'] = df_cat['class'].astype('category')
        logger.info(f"class 列已转换为分类变量，类别: {list(df_cat['class'].cat.categories)}")
    
    return df_cat

def calculate_survival_rates(df):
    """计算生存率"""
    logger.info("开始计算生存率...")
    
    # 按性别计算生存率
    if 'sex' in df.columns and 'survived' in df.columns:
        gender_survival = df.groupby('sex')['survived'].agg(['count', 'mean'])
        gender_survival['survival_rate'] = gender_survival['mean'] * 100
        logger.info("按性别生存率:")
        for gender in gender_survival.index:
            rate = gender_survival.loc[gender, 'survival_rate']
            count = gender_survival.loc[gender, 'count']
            logger.info(f"  {gender}: {rate:.1f}% ({int(gender_survival.loc[gender, 'mean'] * count)}/{count})")
    
    # 按舱位等级计算生存率
    if 'pclass' in df.columns and 'survived' in df.columns:
        class_survival = df.groupby('pclass')['survived'].agg(['count', 'mean'])
        class_survival['survival_rate'] = class_survival['mean'] * 100
        logger.info("按舱位等级生存率:")
        for pclass in class_survival.index:
            rate = class_survival.loc[pclass, 'survival_rate']
            count = class_survival.loc[pclass, 'count']
            logger.info(f"  {pclass}等舱: {rate:.1f}% ({int(class_survival.loc[pclass, 'mean'] * count)}/{count})")
    
    # 按性别和舱位等级组合计算生存率
    if all(col in df.columns for col in ['sex', 'pclass', 'survived']):
        combined_survival = df.groupby(['sex', 'pclass'])['survived'].agg(['count', 'mean'])
        combined_survival['survival_rate'] = combined_survival['mean'] * 100
        logger.info("按性别和舱位等级组合生存率:")
        for (sex, pclass), row in combined_survival.iterrows():
            rate = row['survival_rate']
            count = row['count']
            logger.info(f"  {sex}-{pclass}等舱: {rate:.1f}% ({int(row['mean'] * count)}/{count})")
    
    return gender_survival, class_survival, combined_survival if 'combined_survival' in locals() else None

def create_is_alone_feature(df):
    """创建 is_alone 特征"""
    logger.info("开始创建 is_alone 特征...")
    df_alone = df.copy()
    
    if all(col in df_alone.columns for col in ['sibsp', 'parch']):
        df_alone['is_alone'] = (df_alone['sibsp'] + df_alone['parch']) == 0
        df_alone['is_alone'] = df_alone['is_alone'].astype(bool)
        
        alone_count = df_alone['is_alone'].sum()
        not_alone_count = len(df_alone) - alone_count
        logger.info(f"is_alone 特征创建完成: {alone_count} 人独自旅行，{not_alone_count} 人有家人同行")
        
        # 计算独自旅行者的生存率
        if 'survived' in df_alone.columns:
            alone_survival = df_alone.groupby('is_alone')['survived'].mean() * 100
            for alone, rate in alone_survival.items():
                status = "独自" if alone else "有家人"
                logger.info(f"  {status}旅行生存率: {rate:.1f}%")
    
    return df_alone
