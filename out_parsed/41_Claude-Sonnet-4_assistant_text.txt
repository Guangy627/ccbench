import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import json
import numpy as np

# 读取数据
df = pd.read_csv('ml_datasets/gapminderDataFiveYear.csv')

# 创建输出目录（如果需要）
import os
os.makedirs('plots', exist_ok=True)

# 任务1: 筛选2007年数据并绘制散点图
data_2007 = df[df['year'] == 2007].copy()

plt.figure(figsize=(12, 8))
colors = {'Asia': 'red', 'Europe': 'blue', 'Africa': 'green', 'Americas': 'orange', 'Oceania': 'purple'}
for continent in data_2007['continent'].unique():
    subset = data_2007[data_2007['continent'] == continent]
    plt.scatter(subset['gdpPercap'], subset['lifeExp'], 
               label=continent, color=colors.get(continent, 'black'), alpha=0.7, s=60)

plt.xlabel('GDP per Capita')
plt.ylabel('Life Expectancy')
plt.title('GDP per Capita vs Life Expectancy in 2007 by Continent')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('scatter_plot_2007.png', dpi=300, bbox_inches='tight')
plt.close()

# 任务2: 计算各大洲平均人口并绘制柱状图
continent_pop = data_2007.groupby('continent')['pop'].mean().sort_values(ascending=False)

plt.figure(figsize=(10, 6))
bars = plt.bar(continent_pop.index, continent_pop.values, 
               color=['red', 'blue', 'green', 'orange', 'purple'])
plt.xlabel('Continent')
plt.ylabel('Average Population')
plt.title('Average Population by Continent in 2007')
plt.xticks(rotation=45)
plt.grid(True, alpha=0.3, axis='y')

# 添加数值标签
for bar, value in zip(bars, continent_pop.values):
    plt.text(bar.get_x() + bar.get_width()/2, bar.get_height() + value*0.01,
             f'{value:,.0f}', ha='center', va='bottom')

plt.tight_layout()
plt.savefig('population_by_continent_2007.png', dpi=300, bbox_inches='tight')
plt.close()

# 任务3: 找出人口增长最快的国家（1952→2007）
data_1952 = df[df['year'] == 1952][['country', 'pop']].rename(columns={'pop': 'pop_1952'})
data_2007_pop = df[df['year'] == 2007][['country', 'pop']].rename(columns={'pop': 'pop_2007'})

population_growth = pd.merge(data_1952, data_2007_pop, on='country')
population_growth['growth_rate'] = (population_growth['pop_2007'] - population_growth['pop_1952']) / population_growth['pop_1952'] * 100
population_growth = population_growth.sort_values('growth_rate', ascending=False)

fastest_growing = population_growth.head(10)

plt.figure(figsize=(12, 8))
plt.barh(range(len(fastest_growing)), fastest_growing['growth_rate'], 
         color='skyblue', edgecolor='navy')
plt.yticks(range(len(fastest_growing)), fastest_growing['country'])
plt.xlabel('Population Growth Rate (%)')
plt.title('Top 10 Countries with Fastest Population Growth (1952-2007)')
plt.grid(True, alpha=0.3, axis='x')

# 添加数值标签
for i, value in enumerate(fastest_growing['growth_rate']):
    plt.text(value + 5, i, f'{value:.1f}%', va='center')

plt.tight_layout()
plt.savefig('population_growth_1952_2007.png', dpi=300, bbox_inches='tight')
plt.close()

# 任务4: 创建gdp_per_capita新列（实际上数据中的gdpPercap就是人均GDP）
# 但我们创建一个总GDP列作为演示
df_with_gdp = df.copy()
df_with_gdp['total_gdp'] = df_with_gdp['gdpPercap'] * df_with_gdp['pop']
df_with_gdp['gdp_per_capita_calculated'] = df_with_gdp['total_gdp'] / df_with_gdp['pop']

# 任务5: 将2007年数据导出为JSON
data_2007_with_gdp = df_with_gdp[df_with_gdp['year'] == 2007].copy()
data_2007_json = data_2007_with_gdp.to_dict(orient='records')

with open('gapminder_2007.json', 'w', encoding='utf-8') as f:
    json.dump(data_2007_json, f, indent=2, ensure_ascii=False)

# 生成分析结果摘要
results = {
    'analysis_summary': {
        'total_countries_2007': len(data_2007),
        'continents': list(data_2007['continent'].unique()),
        'avg_life_expectancy_2007': float(data_2007['lifeExp'].mean()),
        'avg_gdp_per_capita_2007': float(data_2007['gdpPercap'].mean()),
        'continent_avg_population': continent_pop.to_dict(),
        'fastest_growing_country': {
            'country': fastest_growing.iloc[0]['country'],
            'growth_rate_percent': float(fastest_growing.iloc[0]['growth_rate'])
        },
        'top_10_fastest_growing': fastest_growing[['country', 'growth_rate']].to_dict(orient='records')
    }
}

with open('analysis_results.json', 'w', encoding='utf-8') as f:
    json.dump(results, f, indent=2, ensure_ascii=False)

print("分析完成！生成的文件：")
print("- scatter_plot_2007.png: 2007年GDP与预期寿命散点图")
print("- population_by_continent_2007.png: 2007年各大洲平均人口柱状图")
print("- population_growth_1952_2007.png: 人口增长最快的国家")
print("- gapminder_2007.json: 2007年数据JSON文件")
print("- analysis_results.json: 分析结果摘要")
