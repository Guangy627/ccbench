#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Spotify歌曲数据分析脚本
完成以下分析任务：
1. 提取年份列并筛选2000年及以后的数据
2. 分析每年歌曲数量趋势
3. 计算活跃度得分并找出前5首歌曲
4. 比较不同genre的danceability差异
5. 绘制tempo分布直方图

import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime
import numpy as np

# 设置中文字体支持
plt.rcParams['font.sans-serif'] = ['DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

# 读取数据
def load_data():
    """加载Spotify歌曲数据"""
    df = pd.read_csv('ml_datasets/spotify_songs.csv')
    return df

def extract_year_and_filter(df):
    """提取年份列并筛选2000年及以后的数据"""
    # 提取年份
    df['release_year'] = pd.to_datetime(df['track_album_release_date']).dt.year
    
    # 筛选2000年及以后的数据
    df_filtered = df[df['release_year'] >= 2000].copy()
    
    print(f"原始数据量: {len(df)}")
    print(f"筛选后数据量: {len(df_filtered)}")
    print(f"年份范围: {df_filtered['release_year'].min()} - {df_filtered['release_year'].max()}")
    
    return df_filtered

def analyze_yearly_trends(df):
    """分析每年歌曲数量趋势"""
    yearly_counts = df['release_year'].value_counts().sort_index()
    
    # 绘制年度趋势图
    plt.figure(figsize=(12, 6))
    plt.plot(yearly_counts.index, yearly_counts.values, marker='o', linewidth=2, markersize=6)
    plt.title('Spotify歌曲年度发布趋势 (2000年及以后)', fontsize=14, fontweight='bold')
    plt.xlabel('年份', fontsize=12)
    plt.ylabel('歌曲数量', fontsize=12)
    plt.grid(True, alpha=0.3)
    plt.xticks(rotation=45)
    plt.tight_layout()
    plt.savefig('yearly_trend.png', dpi=300, bbox_inches='tight')
    plt.close()
    
    return yearly_counts

def calculate_activity_score(df):
    """计算活跃度得分并找出前5首歌曲"""
    df['activity_score'] = df['energy'] * 0.5 + df['valence'] * 0.5
    
    # 找出得分最高的5首歌曲
    top_songs = df.nlargest(5, 'activity_score')[['track_name', 'track_artist', 'activity_score', 'energy', 'valence']]
    
    # 绘制前5首歌曲的得分图
    plt.figure(figsize=(10, 6))
    songs_labels = [f"{row['track_name'][:20]}...\n({row['track_artist'][:15]}...)" 
                   for _, row in top_songs.iterrows()]
    
    x = range(len(top_songs))
    width = 0.35
    
    plt.bar([i - width/2 for i in x], top_songs['energy'], width, label='Energy', alpha=0.7)
    plt.bar([i + width/2 for i in x], top_songs['valence'], width, label='Valence', alpha=0.7)
    plt.bar(x, top_songs['activity_score'], width=0.1, color='red', label='Activity Score')
    
    plt.title('活跃度得分最高的5首歌曲', fontsize=14, fontweight='bold')
    plt.xlabel('歌曲', fontsize=12)
    plt.ylabel('得分', fontsize=12)
    plt.xticks(x, songs_labels, rotation=45, ha='right')
    plt.legend()
    plt.tight_layout()
    plt.savefig('top_songs_activity.png', dpi=300, bbox_inches='tight')
    plt.close()
    
    return top_songs

def compare_genre_danceability(df):
    """比较不同genre的danceability差异"""
    genre_danceability = df.groupby('playlist_genre')['danceability'].mean().sort_values(ascending=False)
    
    # 绘制柱状图
    plt.figure(figsize=(10, 6))
    colors = plt.cm.Set3(np.linspace(0, 1, len(genre_danceability)))
    bars = plt.bar(genre_danceability.index, genre_danceability.values, color=colors)
    
    # 添加数值标签
    for i, v in enumerate(genre_danceability.values):
        plt.text(i, v + 0.005, f'{v:.3f}', ha='center', va='bottom')
    
    plt.title('不同音乐类型的平均Danceability比较', fontsize=14, fontweight='bold')
    plt.xlabel('音乐类型', fontsize=12)
    plt.ylabel('平均Danceability', fontsize=12)
    plt.xticks(rotation=45)
    plt.ylim(0, max(genre_danceability.values) * 1.1)
    plt.tight_layout()
    plt.savefig('genre_danceability.png', dpi=300, bbox_inches='tight')
    plt.close()
    
    return genre_danceability

def plot_tempo_distribution(df):
    """绘制tempo分布直方图"""
    tempo_data = df['tempo']
    mean_tempo = tempo_data.mean()
    median_tempo = tempo_data.median()
    
    plt.figure(figsize=(10, 6))
    plt.hist(tempo_data, bins=30, alpha=0.7, color='skyblue', edgecolor='black')
    
    # 标注均值和中位数
    plt.axvline(mean_tempo, color='red', linestyle='--', linewidth=2, label=f'均值: {mean_tempo:.1f} BPM')
    plt.axvline(median_tempo, color='green', linestyle='--', linewidth=2, label=f'中位数: {median_tempo:.1f} BPM')
    
    plt.title('Tempo (BPM) 分布直方图', fontsize=14, fontweight='bold')
    plt.xlabel('Tempo (BPM)', fontsize=12)
    plt.ylabel('歌曲数量', fontsize=12)
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.savefig('tempo_distribution.png', dpi=300, bbox_inches='tight')
    plt.close()
    
    return mean_tempo, median_tempo

def extract_year_and_filter(df):
    """提取年份列并筛选2000年及以后的数据"""
    # 处理多种日期格式
    def extract_year(date_str):
        if pd.isna(date_str):
            return None
        
        # 如果只有年份（4位数字）
        if len(str(date_str)) == 4 and str(date_str).isdigit():
            year = int(date_str)
            return year if year >= 1000 else None
        
        # 如果是年月格式（如1981-12）
        if len(str(date_str)) == 7 and '-' in str(date_str):
            try:
                year_part = str(date_str).split('-')[0]
                if year_part.isdigit():
                    year = int(year_part)
                    return year if year >= 1000 else None
            except:
                pass
        
        # 尝试标准日期解析
        try:
            return pd.to_datetime(date_str, errors='coerce').year
        except:
            return None
    
    df['release_year'] = df['track_album_release_date'].apply(extract_year)
    
    # 筛选2000年及以后的数据
    df_filtered = df[df['release_year'] >= 2000].copy()
    
    # 确保年份是整数
    df_filtered['release_year'] = df_filtered['release_year'].astype(int)
    
    print(f"原始数据量: {len(df)}")
    print(f"筛选后数据量: {len(df_filtered)}")
    print(f"年份范围: {df_filtered['release_year'].min()} - {df_filtered['release_year'].max()}")
    
    return df_filtered
