#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
数据清洗和分析脚本
处理Boston Housing和Titanic数据集
"""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats
import logging
import os
from datetime import datetime

# 设置日志
logging.basicConfig(
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('data_analysis.log'),
        logging.StreamHandler()
    ]
)
logger = logging.getLogger(__name__)

# 设置中文字体
plt.rcParams['font.sans-serif'] = ['SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

class DataAnalyzer:
    def __init__(self):
        self.results = {}
        
    def analyze_boston_housing(self):
        """分析Boston Housing数据集"""
        logger.info("开始分析Boston Housing数据集...")
        
        try:
            # 读取数据
            df = pd.read_csv('ml_datasets/BostonHousing.txt', sep='\s+')
            logger.info(f"Boston Housing数据加载成功，形状: {df.shape}")
            
            # 任务1: 检查medv异常值(IQR方法)
            logger.info("检查medv列的异常值...")
            Q1 = df['medv'].quantile(0.25)
            Q3 = df['medv'].quantile(0.75)
            IQR = Q3 - Q1
            lower_bound = Q1 - 1.5 * IQR
            upper_bound = Q3 + 1.5 * IQR
            
            outliers = df[(df['medv'] < lower_bound) | (df['medv'] > upper_bound)]
            logger.info(f"发现异常值数量: {len(outliers)}")
            logger.info(f"异常值范围: < {lower_bound:.2f} 或 > {upper_bound:.2f}")
            
            # 任务2: 找出与medv相关性最强的3个特征
            logger.info("计算与medv的皮尔逊相关系数...")
            correlations = df.corr()['medv'].sort_values(ascending=False)
            top_3_features = correlations[1:4]  # 排除medv自身
            
            logger.info("与medv相关性最强的3个特征:")
            for feature, corr in top_3_features.items():
                logger.info(f"  {feature}: {corr:.3f}")
            
            # 任务3: 创建rm_rounded列并计算平均房价
            logger.info("创建rm_rounded列...")
            df['rm_rounded'] = df['rm'].round()
            avg_price_by_rooms = df.groupby('rm_rounded')['medv'].mean()
            
            logger.info("不同房间数的平均房价:")
            for rooms, avg_price in avg_price_by_rooms.items():
                logger.info(f"  {rooms}个房间: ${avg_price:.2f}")
            
            # 任务4: 绘制rm与medv的散点图
            logger.info("绘制rm与medv的散点图...")
            plt.figure(figsize=(10, 6))
            sns.scatterplot(data=df, x='rm', y='medv', alpha=0.6)
            
            # 添加趋势线
            z = np.polyfit(df['rm'], df['medv'], 1)
            p = np.poly1d(z)
            plt.plot(df['rm'], p(df['rm']), "r--", alpha=0.8)
            
            plt.title('房间数量(rm)与房价(medv)的关系')
            plt.xlabel('房间数量 (rm)')
            plt.ylabel('房价 (medv)')
            plt.grid(True, alpha=0.3)
            plt.tight_layout()
            plt.savefig('boston_rm_medv_scatter.png', dpi=300, bbox_inches='tight')
            plt.close()
            
            # 任务5: 保存清洗后的数据
            logger.info("保存清洗后的数据...")
            df_clean = df[(df['medv'] >= lower_bound) & (df['medv'] <= upper_bound)]
            df_clean.to_csv('boston_clean.csv', index=False)
            logger.info(f"清洗后数据形状: {df_clean.shape}")
            
            # 保存结果
            self.results['boston'] = {
                'original_shape': df.shape,
                'clean_shape': df_clean.shape,
                'outliers_count': len(outliers),
                'correlations': top_3_features.to_dict(),
                'avg_price_by_rooms': avg_price_by_rooms.to_dict(),
                'scatter_plot': 'boston_rm_medv_scatter.png'
            }
            
            logger.info("Boston Housing数据分析完成!")
            
        except Exception as e:
            logger.error(f"Boston Housing分析出错: {e}")
            raise
    
    def analyze_titanic(self):
        """分析Titanic数据集"""
        logger.info("开始分析Titanic数据集...")
        
        try:
            # 读取数据
            df = pd.read_csv('ml_datasets/titanic.csv')
            logger.info(f"Titanic数据加载成功，形状: {df.shape}")
            
            # 任务1: 检查缺失值
            logger.info("检查缺失值...")
            missing_data = df.isnull().sum()
            logger.info("缺失值统计:")
            for col, count in missing_data.items():
                if count > 0:
                    logger.info(f"  {col}: {count}个缺失值 ({count/len(df)*100:.1f}%)")
            
            # 填充缺失值
            logger.info("填充缺失值...")
            
            # age列用中位数填充
            age_median = df['age'].median()
            df['age'].fillna(age_median, inplace=True)
            logger.info(f"age列用中位数填充: {age_median:.1f}")
            
            # deck列用众数填充
            deck_mode = df['deck'].mode()[0] if not df['deck'].mode().empty else 'Unknown'
            df['deck'].fillna(deck_mode, inplace=True)
            logger.info(f"deck列用众数填充: {deck_mode}")
            
            # 任务2: 转换分类变量
            logger.info("转换分类变量...")
            df['embarked'] = df['embarked'].astype('category')
            df['class'] = df['class'].astype('category')
            logger.info("embarked和class列已转换为category类型")
            
            # 任务3: 计算不同性别、舱位等级的生存率
            logger.info("计算生存率...")
            
            # 按性别计算生存率
            survival_by_sex = df.groupby('sex')['survived'].mean() * 100
            logger.info("按性别生存率:")
            for sex, rate in survival_by_sex.items():
                logger.info(f"  {sex}: {rate:.1f}%")
            
            # 按舱位等级计算生存率
            survival_by_class = df.groupby('class')['survived'].mean() * 100
            logger.info("按舱位等级生存率:")
            for class_, rate in survival_by_class.items():
                logger.info(f"  {class_}: {rate:.1f}%")
            
            # 按性别和舱位等级组合计算生存率
            survival_by_sex_class = df.groupby(['sex', 'class'])['survived'].mean() * 100
            logger.info("按性别和舱位等级组合生存率:")
            for (sex, class_), rate in survival_by_sex_class.items():
                logger.info(f"  {sex}-{class_}: {rate:.1f}%")
            
            # 绘制生存率柱状图
            logger.info("绘制生存率柱状图...")
            fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6))
            
            # 按性别生存率
            survival_by_sex.plot(kind='bar', ax=ax1, color=['lightcoral', 'lightblue'])
            ax1.set_title('按性别生存率')
            ax1.set_ylabel('生存率 (%)')
            ax1.set_ylim(0, 100)
            
            # 按舱位等级生存率
            survival_by_class.plot(kind='bar', ax=ax2, color=['gold', 'lightgreen', 'lightcoral'])
            ax2.set_title('按舱位等级生存率')
            ax2.set_ylabel('生存率 (%)')
            ax2.set_ylim(0, 100)
            
            plt.tight_layout()
            plt.savefig('titanic_survival_rates.png', dpi=300, bbox_inches='tight')
            plt.close()
            
            # 任务4: 创建is_alone特征
            logger.info("创建is_alone特征...")
            df['is_alone'] = (df['sibsp'] + df['parch']) == 0
            alone_survival = df.groupby('is_alone')['survived'].mean() * 100
            
            logger.info("独自旅行与有家人陪伴的生存率:")
            for alone, rate in alone_survival.items():
                status = "独自旅行" if alone else "有家人陪伴"
                logger.info(f"  {status}: {rate:.1f}%")
            
            # 任务5: 保存清洗后的数据
            logger.info("保存清洗后的数据...")
            df.to_csv('titanic_clean.csv', index=False)
            logger.info(f"清洗后数据形状: {df.shape}")
            
            # 保存结果
            self.results['titanic'] = {
                'original_shape': df.shape,
                'clean_shape': df.shape,  # 没有删除行，只是填充了缺失值
                'missing_data': missing_data[missing_data > 0].to_dict(),
                'survival_by_sex': survival_by_sex.to_dict(),
                'survival_by_class': survival_by_class.to_dict(),
                'survival_by_sex_class': survival_by_sex_class.to_dict(),
                'alone_survival': alone_survival.to_dict(),
                'survival_plot': 'titanic_survival_rates.png'
            }
            
            logger.info("Titanic数据分析完成!")
            
        except Exception as e:
            logger.error(f"Titanic分析出错: {e}")
            raise
