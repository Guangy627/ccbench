import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
from io import StringIO

# 设置中文字体支持
plt.rcParams['font.sans-serif'] = ['SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

# 读取数据
df = pd.read_csv('/app/projects/ml_datasets/diamonds.csv')

# 创建结果存储字典
results = {}

# 1. 查看cut、color、clarity的类别分布
results['cut_distribution'] = df['cut'].value_counts()
results['color_distribution'] = df['color'].value_counts()
results['clarity_distribution'] = df['clarity'].value_counts()

# 2. 计算每种cut类型的平均价格和中位价格
price_stats = df.groupby('cut')['price'].agg(['mean', 'median']).round(2)
results['cut_price_stats'] = price_stats

# 3. 找出价格最高的10颗钻石，并查看它们的carat和z（深度）
top_10_diamonds = df.nlargest(10, 'price')[['carat', 'z', 'price', 'cut', 'color', 'clarity']]
results['top_10_diamonds'] = top_10_diamonds

# 4. 绘制carat与price的散点图（抽样1000条）
if len(df) > 1000:
    sample_df = df.sample(1000, random_state=42)
else:
    sample_df = df

plt.figure(figsize=(10, 6))
plt.scatter(sample_df['carat'], sample_df['price'], alpha=0.6)
plt.xlabel('Carat')
plt.ylabel('Price')
plt.title('Carat vs Price (1000样本抽样)')
plt.grid(True, alpha=0.3)
plt.savefig('/app/projects/carat_price_scatter.png', dpi=300, bbox_inches='tight')
plt.close()

# 5. 创建价格分档列price_bin，并统计各档数量
bins = [0, 1000, 5000, 10000, 15000, 20000, df['price'].max()]
labels = ['0-1000', '1000-5000', '5000-10000', '10000-15000', '15000-20000', '20000+']
df['price_bin'] = pd.cut(df['price'], bins=bins, labels=labels, right=False)
price_bin_counts = df['price_bin'].value_counts().sort_index()
results['price_bin_counts'] = price_bin_counts

# 生成HTML报告
html_content = f"""
