import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
from datetime import datetime
import warnings
warnings.filterwarnings('ignore')

# Read the dataset
df = pd.read_csv('ml_datasets/spotify_songs.csv')

# Extract release year and filter data from 2000+
df['track_album_release_date'] = pd.to_datetime(df['track_album_release_date'])
df['release_year'] = df['track_album_release_date'].dt.year
df_filtered = df[df['release_year'] >= 2000].copy()

print(f"Original dataset: {len(df)} songs")
print(f"Filtered dataset (2000+): {len(df_filtered)} songs")

# Set up plotting style
plt.style.use('default')
sns.set_palette("husl")

# 1. Yearly song release trends
plt.figure(figsize=(12, 6))
yearly_counts = df_filtered.groupby('release_year').size()
plt.plot(yearly_counts.index, yearly_counts.values, marker='o', linewidth=2, markersize=4)
plt.title('Annual Song Release Trends (2000+)', fontsize=16, fontweight='bold')
plt.xlabel('Year', fontsize=12)
plt.ylabel('Number of Songs Released', fontsize=12)
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('yearly_trends.png', dpi=300, bbox_inches='tight')
plt.close()

# 2. Calculate activity score and find top 5 songs
df_filtered['activity_score'] = df_filtered['energy'] * 0.5 + df_filtered['valence'] * 0.5
top_5_songs = df_filtered.nlargest(5, 'activity_score')[['track_name', 'track_artist', 'activity_score']]

print("\nTop 5 Songs by Activity Score:")
for idx, row in top_5_songs.iterrows():
    print(f"{row['track_name']} - {row['track_artist']} (Score: {row['activity_score']:.3f})")

# 3. Compare danceability by genre
plt.figure(figsize=(12, 8))
genre_dance = df_filtered.groupby('playlist_genre')['danceability'].mean().sort_values(ascending=True)
bars = plt.barh(range(len(genre_dance)), genre_dance.values, color=sns.color_palette("viridis", len(genre_dance)))
plt.yticks(range(len(genre_dance)), genre_dance.index)
plt.xlabel('Average Danceability', fontsize=12)
plt.title('Average Danceability by Genre', fontsize=16, fontweight='bold')
plt.grid(True, alpha=0.3, axis='x')

# Add value labels on bars
for i, v in enumerate(genre_dance.values):
    plt.text(v + 0.01, i, f'{v:.3f}', va='center', fontsize=10)

plt.tight_layout()
plt.savefig('danceability_by_genre.png', dpi=300, bbox_inches='tight')
plt.close()

# 4. Tempo distribution histogram
plt.figure(figsize=(12, 6))
tempo_mean = df_filtered['tempo'].mean()
tempo_median = df_filtered['tempo'].median()

plt.hist(df_filtered['tempo'], bins=50, alpha=0.7, color='skyblue', edgecolor='black')
plt.axvline(tempo_mean, color='red', linestyle='--', linewidth=2, label=f'Mean: {tempo_mean:.1f} BPM')
plt.axvline(tempo_median, color='orange', linestyle='--', linewidth=2, label=f'Median: {tempo_median:.1f} BPM')
plt.xlabel('Tempo (BPM)', fontsize=12)
plt.ylabel('Frequency', fontsize=12)
plt.title('Distribution of Tempo (BPM)', fontsize=16, fontweight='bold')
plt.legend()
plt.grid(True, alpha=0.3)
plt.tight_layout()
plt.savefig('tempo_distribution.png', dpi=300, bbox_inches='tight')
plt.close()

# Generate summary statistics for report
stats = {
    'total_songs': len(df_filtered),
    'year_range': f"{df_filtered['release_year'].min()}-{df_filtered['release_year'].max()}",
    'avg_tempo': tempo_mean,
    'median_tempo': tempo_median,
    'top_genre_danceability': genre_dance.index[-1],
    'max_danceability': genre_dance.values[-1],
    'yearly_peak': yearly_counts.idxmax(),
    'yearly_peak_count': yearly_counts.max()
}

# Save analysis results
results = {
    'stats': stats,
    'top_songs': top_5_songs.to_dict('records'),
    'genre_danceability': genre_dance.to_dict(),
    'yearly_trends': yearly_counts.to_dict()
}

import json
with open('analysis_results.json', 'w') as f:
    json.dump(results, f, indent=2, default=str)

print("\nAnalysis complete! Files generated:")
print("- yearly_trends.png")
print("- danceability_by_genre.png")
print("- tempo_distribution.png")
print("- analysis_results.json")
